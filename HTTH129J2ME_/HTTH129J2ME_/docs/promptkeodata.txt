    private static final String DEFAULT_SCHEMA = "pirvateserverz";
    private static final String DEFAULT_TABLE = "effect_data";
    private static final Path BASE_RES = Paths.get("C:", "ThMadara", "HTTH", "res", "effect");
    private static final Path BASE_SQL = Paths.get("C:", "ThMadara", "HTTH", "SQL", "eff");
    private static final Path BASE_RAW = Paths.get("C:", "ThMadara", "HTTH", "data", "template", "skill");
    private static final int TIMEOUT_MS = 3000;
    private static final int RETRIES = 1;

    public static void startRange(int fromId, int toId, int threads, int zoom, int pauseMs) {
        if (fromId < 0) fromId = 0;
        if (toId < fromId) return;
        try {
            Files.createDirectories(BASE_RES.resolve(String.valueOf(zoom)));
            Files.createDirectories(BASE_SQL);
            Files.createDirectories(BASE_RAW);
        } catch (IOException e) {
            e.printStackTrace();
            return;
        }
        ExecutorService ex = Executors.newFixedThreadPool(Math.max(1, threads));
        List<Future<?>> futures = new ArrayList<>();
        for (int id = fromId; id <= toId; id++) {
            final short eff = (short) id;
            futures.add(ex.submit(() -> {
                byte[] raw = null;
                byte[] img = null;
                for (int attempt = 0; attempt <= RETRIES; attempt++) {
                    try {
                        sendRequestForEffect(eff);
                        Wrapper w = waitForEffectData(eff);
                        if (w != null && w.data != null) {
                            raw = w.data;
                            img = w.img;
                            DataSkillEff.ALL_EFF_DATA.remove(String.valueOf(eff));
                            break;
                        }
                    } catch (Throwable t) {}
                    try { Thread.sleep(pauseMs); } catch (InterruptedException ie) { return; }
                }
                ParsedEffect parsed = raw != null ? parse(raw) : null;
                try {
                    saveFilesAtomic(eff, raw, img, zoom, parsed);
                    System.out.println("[OK] effId=" + eff);
                } catch (IOException e) {
                    System.err.println("[FAIL SAVE] effId=" + eff + ": " + e.getMessage());
                }
            }));
        }
        ex.shutdown();
        try {
            ex.awaitTermination(1, TimeUnit.HOURS);
        } catch (InterruptedException ignored) {}
        System.out.println("Done " + fromId + "-" + toId);
    }

    private static void sendRequestForEffect(short effId) {
        try {
            GlobalService.getInstance().getDataSkillEFf((byte)0, effId);
        } catch (Throwable t) {
            try {
                Message m = new Message((byte)74);
                DataOutputStream dos = m.writer();
                dos.writeByte(0);
                dos.writeShort(effId);
                dos.flush();
                Session_ME.getInstance().AA(m);
            } catch (Throwable ignored) {}
        }
    }

    private static Wrapper waitForEffectData(short effId) {
        long start = System.currentTimeMillis();
        String key = String.valueOf(effId);
        while (System.currentTimeMillis() - start < TIMEOUT_MS) {
            try {
                Object o = DataSkillEff.ALL_EFF_DATA.get(key);
                if (o != null) {
                    byte[] data = (byte[]) o.getClass().getField("data").get(o);
                    byte[] imgRaw = (byte[]) o.getClass().getField("imgRaw").get(o);
                    if (data != null) return new Wrapper(data, imgRaw);
                }
            } catch (Throwable ignored) {}
            try { Thread.sleep(20); } catch (InterruptedException ignored) { return null; }
        }
        return null;
    }

    private static class Wrapper {
        byte[] data, img;
        Wrapper(byte[] d, byte[] i) { data = d; img = i; }
    }

    private static void saveFilesAtomic(short effId, byte[] raw, byte[] img, int zoom, ParsedEffect parsed) throws IOException {
        Path zoomDir = BASE_RES.resolve(String.valueOf(zoom));
        if (img != null && img.length > 8) {
            String ext = detectImageExt(img);
            atomicWrite(zoomDir.resolve(effId + ext), img);
        }
        if (raw != null && raw.length > 0) {
            atomicWrite(BASE_RAW.resolve(effId + ".data"), raw);
        }
        String sql = parsed != null ? generateUpdateSQL(DEFAULT_SCHEMA, DEFAULT_TABLE, effId, parsed) : generateFailSql(effId);
        atomicWrite(BASE_SQL.resolve("effect_" + effId + ".sql"), sql.getBytes("UTF-8"));
    }

    private static void atomicWrite(Path file, byte[] data) throws IOException {
        Path tmp = file.getParent().resolve(file.getFileName() + ".tmp");
        Files.write(tmp, data);
        Files.move(tmp, file, StandardCopyOption.REPLACE_EXISTING);
    }

    public static String generateUpdateSQL(String schema, String table, int effId, ParsedEffect parsed) {
        if (schema == null) schema = DEFAULT_SCHEMA;
        if (table == null) table = DEFAULT_TABLE;
        int numImages = parsed != null && parsed.smallImages != null ? parsed.smallImages.length : 0;
        int seqLength = parsed != null && parsed.sequence != null ? parsed.sequence.length : 0;
        int hasExtended = parsed != null && parsed.charFrames != null ? 1 : 0;
        String images = escapeJsonForSql(buildImagesJson(parsed != null ? parsed.smallImages : null));
        String frames = escapeJsonForSql(buildFramesJson(parsed != null ? parsed.frames : null));
        String sequences = escapeJsonForSql(buildSequencesJson(parsed != null ? parsed.sequence : null));
        String charFrames = escapeJsonForSql(buildCharFramesJson(parsed != null ? parsed.charFrames : null));
        String splash = escapeJsonForSql(buildSplashJson(parsed != null ? parsed.indexSplash : null));
        StringBuilder sb = new StringBuilder(2048);
        sb.append("UPDATE `").append(schema).append("`.`").append(table).append("` SET ");
        sb.append("`num_images`=").append(numImages).append(",`seq_length`=").append(seqLength).append(",`has_extended`=").append(hasExtended).append(",");
        sb.append("`images`='").append(images).append("',`frames`='").append(frames).append("',`sequences`='").append(sequences).append("',");
        sb.append("`char_frames`='").append(charFrames).append("',`splash`='").append(splash).append("',`parse_ok`=1,`parse_error`=NULL ");
        sb.append("WHERE `eff_id`=").append(effId).append(";");
        return sb.toString();
    }

    public static String generateFailSql(int effId) {
        return "UPDATE `" + DEFAULT_SCHEMA + "`.`" + DEFAULT_TABLE + "` SET `num_images`=0,`seq_length`=0,`has_extended`=0," +
                "`images`='[]',`frames`='[]',`sequences`='[]',`char_frames`='[]',`splash`='[]',`parse_ok`=0,`parse_error`='timeout_or_missing' " +
                "WHERE `eff_id`=" + effId + ";";
    }

    private static String escapeJsonForSql(String s) {
        return s.replace("\\", "\\\\").replace("'", "\\'").replace("\"", "\\\"");
    }

    private static String detectImageExt(byte[] b) {
        if (b == null || b.length < 4) return ".bin";
        if (b[0] == (byte)0x89 && b[1] == 0x50 && b[2] == 0x4E && b[3] == 0x47) return ".png";
        if (b[0] == 'G' && b[1] == 'I' && b[2] == 'F') return ".gif";
        if (b[0] == (byte)0xFF && b[1] == (byte)0xD8) return ".jpg";
        return ".bin";
    }

    private static String buildImagesJson(SmallImage[] imgs) {
        if (imgs == null || imgs.length == 0) return "[]";
        StringBuilder sb = new StringBuilder("[");
        for (int i = 0; i < imgs.length; i++) {
            SmallImage im = imgs[i];
            sb.append("{\"idx\":").append(i).append(",\"x\":").append(im.x & 0xff).append(",\"y\":").append(im.y & 0xff)
              .append(",\"w\":").append(im.w).append(",\"h\":").append(im.h).append("}");
            if (i < imgs.length - 1) sb.append(",");
        }
        return sb.append("]").toString();
    }

    private static String buildFramesJson(FrameEff[] frames) {
        if (frames == null || frames.length == 0) return "[]";
        StringBuilder sb = new StringBuilder("[");
        for (int fi = 0; fi < frames.length; fi++) {
            FrameEff f = frames[fi];
            sb.append("{\"frame_idx\":").append(fi).append(",\"parts\":[");
            boolean first = true;
            if (f.topParts != null) for (PartFrame p : f.topParts) {
                if (!first) sb.append(",");
                sb.append(partJson(p));
                first = false;
            }
            if (f.bottomParts != null) for (PartFrame p : f.bottomParts) {
                if (!first) sb.append(",");
                sb.append(partJson(p));
                first = false;
            }
            sb.append("]}");
            if (fi < frames.length - 1) sb.append(",");
        }
        return sb.append("]").toString();
    }

    private static String partJson(PartFrame p) {
        return "{\"img_id\":" + p.idSmallImg + ",\"dx\":" + p.dx + ",\"dy\":" + p.dy + ",\"flip\":" + p.flip + ",\"on_top\":" + p.onTop + "}";
    }

    private static String buildSequencesJson(byte[] seq) {
        if (seq == null || seq.length == 0) return "[]";
        StringBuilder sb = new StringBuilder("[");
        for (int i = 0; i < seq.length; i++) {
            sb.append("{\"frame_idx\":").append(seq[i] & 0xff).append("}");
            if (i < seq.length - 1) sb.append(",");
        }
        return sb.append("]").toString();
    }

    private static String buildCharFramesJson(byte[][] charFrames) {
        if (charFrames == null || charFrames.length == 0) return "[]";
        StringBuilder sb = new StringBuilder("[");
        for (int i = 0; i < charFrames.length; i++) {
            byte[] arr = charFrames[i];
            sb.append("[");
            if (arr != null) for (int j = 0; j < arr.length; j++) {
                sb.append(arr[j] & 0xff);
                if (j < arr.length - 1) sb.append(",");
            }
            sb.append("]");
            if (i < charFrames.length - 1) sb.append(",");
        }
        return sb.append("]").toString();
    }

    private static String buildSplashJson(byte[] splash) {
        if (splash == null || splash.length == 0) return "[]";
        StringBuilder sb = new StringBuilder("[");
        for (int i = 0; i < splash.length; i++) {
            sb.append("{\"index_splash\":").append(splash[i] & 0xff).append("}");
            if (i < splash.length - 1) sb.append(",");
        }
        return sb.append("]").toString();
    }

    public static ParsedEffect parse(byte[] raw) {
        if (raw == null || raw.length == 0) return null;
        try (DataInputStream dis = new DataInputStream(new ByteArrayInputStream(raw))) {
            int numImages = dis.readUnsignedByte();
            SmallImage[] smallImages = new SmallImage[numImages];
            for (int i = 0; i < numImages; i++) {
                smallImages[i] = new SmallImage(dis.readUnsignedByte(), dis.readUnsignedByte(), dis.readUnsignedByte(), dis.readUnsignedByte(), dis.readUnsignedByte());
            }
            short numFrames = dis.readShort();
            FrameEff[] frames = new FrameEff[numFrames];
            for (int fi = 0; fi < numFrames; fi++) {
                byte countParts = dis.readByte();
                Vector<PartFrame> topParts = new Vector<>();
                Vector<PartFrame> bottomParts = new Vector<>();
                for (int pi = 0; pi < countParts; pi++) {
                    PartFrame part = new PartFrame(dis.readShort(), dis.readShort(), dis.readByte());
                    part.flip = dis.readByte();
