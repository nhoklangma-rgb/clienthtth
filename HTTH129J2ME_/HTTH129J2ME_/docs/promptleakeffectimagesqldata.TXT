public class EffectService {

    private static final Map<Short, byte[]> imageCache = new HashMap<>();
    private static final String IMAGE_DIR = "data/template/skill";

    public static void sendEffectData(Player p, short effId) {
        byte[] raw = EffectDAO.CACHE_RAW.get(effId);
        if (raw == null) {
            System.err.println("EffectService: raw data for effId=" + effId + " not found!");
            return;
        }
        byte[] img = imageCache.get(effId);
        if (img == null) img = EffectDAO.CACHE_IMG.get(effId);
        if (img == null) {
            img = loadImageFromFile(p.conn.zoomlv,effId);
            if (img != null) {
                imageCache.put(effId, img);
            }
        }
        try {
            Message m = new Message(76);
            DataOutputStream dos = m.writer();
            dos.writeByte(0);
            dos.writeShort(effId);
            if (raw.length > Short.MAX_VALUE) {
                dos.writeShort(Short.MAX_VALUE);
                dos.write(raw, 0, Short.MAX_VALUE);
            } else {
                dos.writeShort(raw.length);
                dos.write(raw);
            }
            if (img != null && img.length > 0) {
                dos.write(img);
            }
            dos.flush();
            p.conn.addmsg(m);
            m.cleanup();
        } catch (IOException e) {
            e.printStackTrace();
        }
    }

 * - Đồng thời cố gắng load image bytes (img) từ DB hoặc từ disk (nhiều path).
 */
public class EffectDAO {

    public static final Map<Short, byte[]> CACHE_RAW = new ConcurrentHashMap<>();
    public static final Map<Short, byte[]> CACHE_IMG = new ConcurrentHashMap<>();
    private static final Gson GSON = new Gson();

    // paths to try for raw data / images (tương đối tới working dir)
    private static final Path[] RAW_PATHS = new Path[] {
            Paths.get("data","template","skill"),
            Paths.get("data","template","skill","x1","data"),
            Paths.get("data","template","skill","x1"),
            Paths.get("data","template","skill","data")

            Paths.get("data","template","skill"),
            Paths.get("data","template","skill","x1"),

            Paths.get("data","template","skill","img")

    public static void load() {
        CACHE_RAW.clear();
        CACHE_IMG.clear();

        String sql = "SELECT * FROM effect_data";
        int rawFromDisk = 0, rawFromDb = 0, rebuilt = 0;
        try (PreparedStatement ps = DbManager.gI().getConnection().prepareStatement(sql);
             ResultSet rs = ps.executeQuery()) {
            while (rs.next()) {
                short effId = rs.getShort("eff_id");
                byte[] raw = null;
                boolean rawFromDiskFlag = false;

                // 1) Try load raw from disk (file names: effId, effId.data, effId.bin, effId.dat)
                raw = tryLoadRawFromDisk(effId);
                if (raw != null) {
                    CACHE_RAW.put(effId, raw);
                    rawFromDisk++;
                    rawFromDiskFlag = true;
                }

                // 2) If no disk raw, try DB raw_data column
                if (!rawFromDiskFlag) {
                    try {
                        byte[] dbRaw = null;
                        try { dbRaw = rs.getBytes("raw_data"); } catch (SQLException ignore) { dbRaw = null; }
                        if (dbRaw != null && dbRaw.length > 0) {
                            int compressed = 0;
                            try { compressed = rs.getInt("raw_compressed"); } catch (SQLException ignore) {}
                            if (compressed == 1) {
                                try { raw = ungzip(dbRaw); } catch (IOException ex) {
                                    Logp.warn("EffectDAO: ungzip failed for effId=" + effId + ", using db raw as-is", ex);
                                    raw = dbRaw;
                                }
                            } else {
                                raw = dbRaw;
                            }
                            if (raw != null && raw.length > 0) {
                                CACHE_RAW.put(effId, raw);
                                rawFromDb++;
                            }
                        }
                    } catch (Throwable t) {
                        Logp.warn("EffectDAO: error reading raw_data for effId=" + effId, t);
                    }
                }

                // 3) If still no raw, try rebuild from JSON (last resort)
                if (!CACHE_RAW.containsKey(effId)) {
                    try {
                        String imagesJson = safeGetString(rs, "images");
                        String framesJson = safeGetString(rs, "frames");
                        String sequencesJson = safeGetString(rs, "sequences");
                        String charFramesJson = safeGetString(rs, "char_frames");
                        String splashJson = safeGetString(rs, "splash");

                        boolean hasJson = (imagesJson != null && !imagesJson.isEmpty()) ||
                                (framesJson != null && !framesJson.isEmpty());
                        if (hasJson) {
                            try {
                                byte[] rebuiltRaw = rebuildRawDataFromParsed(
                                        effId,
                                        rs.getInt("num_images"),
                                        rs.getInt("seq_length"),
                                        rs.getBoolean("has_extended"),
                                        imagesJson,
                                        framesJson,
                                        sequencesJson,
                                        charFramesJson,
                                        splashJson
                                );
                                if (rebuiltRaw != null && rebuiltRaw.length > 0) {
                                    CACHE_RAW.put(effId, rebuiltRaw);
                                    rebuilt++;
                                    Logp.info("EffectDAO: rebuilt raw from JSON effId=" + effId + " len=" + rebuiltRaw.length);
                                }
                            } catch (Exception ex) {
                                Logp.warn("EffectDAO: rebuild fail for effId=" + effId, ex);
                            }
                        }
                    } catch (Throwable t) {
                        Logp.error("EffectDAO: error reading JSON for effId=" + effId, t);
                    }
                }

                // 4) Try load image blob from DB (many possible column names)
                try {
                    byte[] img = null;
                    String[] imgCols = new String[]{"img_data","img","image","image_data","image_blob","img_blob"};
                    for (String c : imgCols) {
                        try {
                            byte[] b = rs.getBytes(c);
                            if (b != null && b.length > 0) { img = b; break; }
                        } catch (SQLException ignore) {}
                    }
                    // If not found in DB, try disk image paths
                    if (img == null) {
                        img = tryLoadImageFromDisk(effId);
                    }
                    if (img != null && img.length > 0) {
                        CACHE_IMG.put(effId, img);
                    }
                } catch (Throwable t) {
                    Logp.warn("EffectDAO: error loading image for effId=" + effId, t);
                }

                if (!CACHE_RAW.containsKey(effId)) {
                    Logp.error("EffectDAO: no raw for effId=" + effId);
                }
            }
            Logp.info("Load effect_data["+CACHE_RAW.size()+"]");
        } catch (SQLException e) {
            Logp.error("EffectDAO: DB error", e);
        }
    }

    private static byte[] tryLoadRawFromDisk(short effId) {
        String[] tryNames = new String[]{"" + effId, effId + ".data", effId + ".dat", effId + ".raw", effId + ".bin"};
        for (Path base : RAW_PATHS) {
            for (String name : tryNames) {
                Path p = base.resolve(name);
                if (Files.exists(p) && Files.isRegularFile(p)) {
                    try {
                        return Files.readAllBytes(p);
                    } catch (IOException ignored) {}
                }
            }
            try (DirectoryStream<Path> ds = Files.newDirectoryStream(base, effId + ".*")) {
                for (Path x : ds) {
                    if (Files.isRegularFile(x)) {
                        try { return Files.readAllBytes(x); } catch (IOException ignored) {}
                    }
                }
            } catch (IOException ignored) {}
        }
        return null;
    }

    }

    private static String safeGetString(ResultSet rs, String col) {
        try {

        } catch (SQLException e) {
            return "";
        }
    }

    private static byte[] ungzip(byte[] gz) throws IOException {
        try (ByteArrayInputStream bais = new ByteArrayInputStream(gz);
             GZIPInputStream gis = new GZIPInputStream(bais);
             ByteArrayOutputStream baos = new ByteArrayOutputStream()) {
            byte[] buf = new byte[8192];
            int r;
            while ((r = gis.read(buf)) != -1) baos.write(buf, 0, r);
            return baos.toByteArray();
        }
    }

    @SuppressWarnings("unchecked")
    private static byte[] rebuildRawDataFromParsed(short effId,
                                                   int numImagesParam,
                                                   int seqLengthParam,
                                                   boolean hasExtendedFromDb,
                                                   String imagesJson,
                                                   String framesJson,
                                                   String sequencesJson,
                                                   String charFramesJson,
                                                   String splashJson) throws IOException {

        Type imageType = new TypeToken<List<Map<String, Number>>>(){}.getType();
        Type framesType = new TypeToken<List<Map<String, Object>>>(){}.getType();
        Type seqType = new TypeToken<List<Map<String, Number>>>(){}.getType();
        Type charType = new TypeToken<List<Map<String, Number>>>(){}.getType();
        Type splashType = new TypeToken<List<Map<String, Number>>>(){}.getType();

        List<Map<String, Number>> images = (imagesJson != null && !imagesJson.isEmpty())
                ? GSON.fromJson(imagesJson, imageType) : Collections.emptyList();
        List<Map<String, Object>> frames = (framesJson != null && !framesJson.isEmpty())
                ? GSON.fromJson(framesJson, framesType) : Collections.emptyList();
        List<Map<String, Number>> sequences = (sequencesJson != null && !sequencesJson.isEmpty())
                ? GSON.fromJson(sequencesJson, seqType) : Collections.emptyList();
        List<Map<String, Number>> charFrames = (charFramesJson != null && !charFramesJson.isEmpty())
                ? GSON.fromJson(charFramesJson, charType) : Collections.emptyList();
        List<Map<String, Number>> splash = (splashJson != null && !splashJson.isEmpty())
                ? GSON.fromJson(splashJson, splashType) : Collections.emptyList();

        int actualNumImages = images.size();
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        DataOutputStream dos = new DataOutputStream(baos);

        dos.writeByte(Math.min(Math.max(actualNumImages, 0), 255));

        for (Map<String, Number> im : images) {
            int x = safeInt(im.get("x"), 0);
            int y = safeInt(im.get("y"), 0);
            int w = Math.max(1, Math.min(safeInt(im.get("w"), 1), 255));
            int h = Math.max(1, Math.min(safeInt(im.get("h"), 1), 255));
            int idx = Math.max(0, Math.min(safeInt(im.get("idx"), 0), 255));
            dos.writeByte(x & 0xFF);
            dos.writeByte(y & 0xFF);
            dos.writeByte(w & 0xFF);
            dos.writeByte(h & 0xFF);
            dos.writeByte(idx & 0xFF);
        }

        int frameCount = frames.size();
        dos.writeShort(frameCount & 0xFFFF);
        dos.writeByte(1);
        for (Map<String, Object> f : frames) {
            List<Map<String, Number>> parts = (List<Map<String, Number>>) f.getOrDefault("parts", Collections.emptyList());
            dos.writeByte(parts.size() & 0xFF);
            for (Map<String, Number> p : parts) {
                int dx = clampShort(safeInt(p.get("dx"), 0));
                int dy = clampShort(safeInt(p.get("dy"), 0));
                int imgId = Math.max(0, Math.min(safeInt(p.get("img_id"), 0), Math.max(0, actualNumImages - 1)));
                int flip = Math.max(0, Math.min(safeInt(p.get("flip"), 0), 255));
                int onTop = Math.max(0, Math.min(safeInt(p.get("on_top"), 0), 255));
                dos.writeShort(dx);
                dos.writeShort(dy);
                dos.writeByte(imgId & 0xFF);
                dos.writeByte(flip & 0xFF);
                dos.writeByte(onTop & 0xFF);
            }
        }

        int seqLen = Math.min(sequences.size(), 255);
        dos.writeByte(seqLen & 0xFF);
        for (Map<String, Number> s : sequences) {
            int frameIdx = safeInt(s.get("frame_idx"), 0);
            if (frameCount > 0) frameIdx = Math.max(0, Math.min(frameIdx, frameCount - 1));
            dos.writeShort(frameIdx & 0xFFFF);
        }

        if (!charFrames.isEmpty()) {
            dos.writeByte(1);
            Map<Integer, List<Integer>> grouped = new HashMap<>();
            for (Map<String, Number> cf : charFrames) {
                int pt = safeInt(cf.get("part_type"), 0);
                int fi = Math.max(0, Math.min(safeInt(cf.get("frame_idx"), 0), 255));
                grouped.computeIfAbsent(pt, k -> new ArrayList<>()).add(fi);
            }
            for (int t = 0; t <= 2; t++) {
                List<Integer> list = grouped.getOrDefault(t, Collections.emptyList());
                dos.writeByte(list.size() & 0xFF);
                for (int v : list) dos.writeByte(v & 0xFF);
            }
        } else {
            dos.writeByte(0);
            dos.writeByte(0);
            dos.writeByte(0);
            dos.writeByte(0);
        }

        for (int i = 0; i < 4; i++) {
            if (i < splash.size()) {
                Map<String, Number> sp = splash.get(i);
                int idx = Math.max(0, Math.min(safeInt(sp.get("index_splash"), 0), 255));
                dos.writeByte(idx & 0xFF);
            } else {
                dos.writeByte(0);
            }
        }

        dos.flush();
        return baos.toByteArray();
    }

    private static int safeInt(Number n, int def) {
        return (n == null) ? def : n.intValue();
    }

    private static int clampShort(int v) {
        return Math.max(Short.MIN_VALUE, Math.min(Short.MAX_VALUE, v));
    }
