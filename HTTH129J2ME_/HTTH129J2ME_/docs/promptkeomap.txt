    /**
     * Load map templates AND build Map[] zones + create Mob instances.
     * IMPORTANT: Call this AFTER MobTemplate.ENTRYS has been populated.
     */
    public static void load() {
        try (Connection conn = DbManager.gI().getConnection();
             PreparedStatement ps = conn.prepareStatement("SELECT * FROM maps");
             ResultSet rs = ps.executeQuery()) {

            ENTRYS = new ArrayList<>();
            if (gameplay.world.maps.Map.ENTRYS == null) {
                gameplay.world.maps.Map.ENTRYS = new ArrayList<>();
            }
//            if (monster.Mob.ENTRYS == null) {
//                monster.Mob.ENTRYS = new HashMap<>();
//            }

            while (rs.next()) {
                short id_map = rs.getShort("id");
                File f = new File("data/map/" + id_map);
                if (!f.exists()) {
                    // skip maps without local map data
                    continue;
                }

                MapTemplate map_temp = new MapTemplate();
                map_temp.id = id_map;
                map_temp.name = rs.getString("name");
                map_temp.max_zone = rs.getByte("maxzone");
                map_temp.max_player = rs.getByte("maxplayer");

                // initialize npcs list now (will be filled later by NpcTemplate.load())
                map_temp.npcs = new ArrayList<>();

                // --- Boat ---
                map_temp.list_boat = new ArrayList<>();
                try {
                    JSONArray js_boat = (JSONArray) JSONValue.parse(rs.getString("boat"));
                    if (js_boat != null) {
                        for (Object obj : js_boat) {
                            JSONArray jsArr = (JSONArray) obj;
                            Boat_In_Map boat = new Boat_In_Map();
                            boat.x = Short.parseShort(jsArr.get(0).toString());
                            boat.y = Short.parseShort(jsArr.get(1).toString());
                            map_temp.list_boat.add(boat);
                        }
                    }
                } catch (Exception ex) {
                    ex.printStackTrace();
                }

                // --- Vgo ---
                map_temp.vgos = new ArrayList<>();
                try {
                    JSONArray js_vgo = (JSONArray) JSONValue.parse(rs.getString("vgos"));
                    if (js_vgo != null) {
                        for (Object obj : js_vgo) {
                            JSONArray jsArr = (JSONArray) obj;
                            Vgo vgo = new Vgo();
                            vgo.id_map_go = Short.parseShort(jsArr.get(0).toString());
                            vgo.xold = Short.parseShort(jsArr.get(1).toString());
                            vgo.yold = Short.parseShort(jsArr.get(2).toString());
                            vgo.xnew = Short.parseShort(jsArr.get(3).toString());
                            vgo.ynew = Short.parseShort(jsArr.get(4).toString());
                            if (vgo.id_map_go != -1) {
                                map_temp.vgos.add(vgo);
                            }
                        }
                    }
                } catch (Exception ex) {
                    ex.printStackTrace();
                }

                // --- data ---
                try {
                    JSONArray js_data = (JSONArray) JSONValue.parse(rs.getString("data"));
                    if (js_data != null) {
                        map_temp.data = new byte[js_data.size()][];
                        for (int i = 0; i < js_data.size(); i++) {
                            JSONArray jsRow = (JSONArray) js_data.get(i);
                            map_temp.data[i] = new byte[jsRow.size()];
                            for (int j = 0; j < jsRow.size(); j++) {
                                map_temp.data[i][j] = Byte.parseByte(jsRow.get(j).toString());
                            }
                        }
                    } else {
                        map_temp.data = new byte[0][0];
                    }
                } catch (Exception ex) {
                    ex.printStackTrace();
                    map_temp.data = new byte[0][0];
                }

                // --- MapBack ---
                try {
                    JSONArray js_back = (JSONArray) JSONValue.parse(rs.getString("MapBack"));
                    if (js_back != null && js_back.size() >= 4) {
                        map_temp.IDBack = Byte.parseByte(js_back.get(0).toString());
                        map_temp.HBack = Short.parseShort(js_back.get(1).toString());
                        map_temp.maxW = Short.parseShort(js_back.get(2).toString());
                        map_temp.maxH = Short.parseShort(js_back.get(3).toString());
                    }
                } catch (Exception ex) {
                    ex.printStackTrace();
                }

                // misc
                map_temp.type_view_p = rs.getByte("typeViewPlayer");
                map_temp.b = rs.getByte("b");
                map_temp.specMap = rs.getByte("specMap");
                map_temp.id_eff_map = rs.getByte("id_eff_map");
                map_temp.level = rs.getByte("level");
                map_temp.typeChangeMap = rs.getByte("typeChangeMap");

                // mPosMapTrain
                try {
                    JSONArray js_train = (JSONArray) JSONValue.parse(rs.getString("mPosMapTrain"));
                    if (js_train != null) {
                        map_temp.mPosMapTrain = new byte[js_train.size()][];
                        for (int i = 0; i < js_train.size(); i++) {
                            JSONArray jsRow = (JSONArray) js_train.get(i);
                            map_temp.mPosMapTrain[i] = new byte[jsRow.size()];
                            for (int j = 0; j < jsRow.size(); j++) {
                                map_temp.mPosMapTrain[i][j] = Byte.parseByte(jsRow.get(j).toString());
                            }
                        }
                    } else {
                        map_temp.mPosMapTrain = new byte[0][0];
                    }
                } catch (Exception ex) {
                    ex.printStackTrace();
                    map_temp.mPosMapTrain = new byte[0][0];
                }

                map_temp.strTimeChange = rs.getString("strTimeChange");

                // --- build Map[] zones and mobs for this template ---
                gameplay.world.maps.Map[] zones = new gameplay.world.maps.Map[Math.max(0, map_temp.max_zone)];
                for (int z = 0; z < zones.length; z++) {
                    zones[z] = new gameplay.world.maps.Map();
                    zones[z].zone_id = (byte) z;
                    zones[z].template = map_temp;
                }

                // parse mobs JSON and create mobs per zone and slot
                try {
                    String mobsJson = rs.getString("mobs");
                    JSONArray js_mobs = null;
                    if (mobsJson != null && !mobsJson.isEmpty()) {
                        js_mobs = (JSONArray) JSONValue.parse(mobsJson);
                    }

                    if (js_mobs == null || js_mobs.size() == 0) {
                        for (int z = 0; z < zones.length; z++) {
                            zones[z].list_mob = new int[0];
                        }
                    } else {
                        for (int z = 0; z < zones.length; z++) {
                            zones[z].list_mob = new int[js_mobs.size()];
                            for (int i = 0; i < js_mobs.size(); i++) {
                                try {
                                    JSONArray js2 = (JSONArray) JSONValue.parse(js_mobs.get(i).toString());
                                    int mobTemplateIndex = Integer.parseInt(js2.get(0).toString());
                                    if (MobTemplate.ENTRYS == null ||
                                            mobTemplateIndex < 0 ||
                                            mobTemplateIndex >= MobTemplate.ENTRYS.size()) {
                                        zones[z].list_mob[i] = -1;
                                        continue;
                                    }
                                    Mob m = new Mob();
                                    m.mob_template = MobTemplate.ENTRYS.get(mobTemplateIndex);
                                    m.x = Short.parseShort(js2.get(1).toString());
                                    m.y = Short.parseShort(js2.get(2).toString());
                                    m.hp_max = m.mob_template.hp_max;
                                    m.hp = m.hp_max;
                                    m.level = m.mob_template.level;
                                    m.isdie = false;
                                    m.id_target = -1;
                                    m.index = index_mob_counter;
                                    m.map = zones[z];
                                    m.boss_info = null;

                                    monster.Mob.ENTRYS.put(index_mob_counter, m);
                                    zones[z].list_mob[i] = index_mob_counter;
                                    index_mob_counter++;
                                } catch (Exception inner) {
                                    inner.printStackTrace();
                                    zones[z].list_mob[i] = -1;
                                }
                            }
                        }
                    }
                } catch (Exception ex) {
                    ex.printStackTrace();
                    for (int z = 0; z < zones.length; z++) {
                        zones[z].list_mob = new int[0];
                    }
                }

                // add into templates and map instances
                ENTRYS.add(map_temp);
                gameplay.world.maps.Map.ENTRYS.add(zones);
            }

            // fix link vgo → map_go (needs Map.ENTRYS populated)
            for (MapTemplate t : ENTRYS) {
                if (t.vgos == null) continue;
                for (Vgo vgo : t.vgos) {
                    vgo.map_go = gameplay.world.maps.Map.get_map_by_id(vgo.id_map_go);
                    if (vgo.map_go == null) {
                        vgo.map_go = gameplay.world.maps.Map.get_map_by_id(1);
                    }
                }
            }
            Logp.info("Load map ["+ENTRYS.size()+"]");
//            System.out.println("Load maps OK, total = " + (ENTRYS != null ? ENTRYS.size() : 0)
//                    + ", mobs index next = " + index_mob_counter);
        } catch (Exception e) {
            e.printStackTrace();
            System.err.println("Error loading maps: " + e.getMessage());
        }
    }

    /**
     * Helper: find MapTemplate by id (used by NpcTemplate)
     */
    public static MapTemplate getById(int mapId) {
        if (ENTRYS == null) return null;
        for (MapTemplate t : ENTRYS) {
            if (t.id == mapId) return t;
        }
        return null;
    }

public void sendDataMap(Player p) {
        try {
            Message m = new Message(0);
            m.writer().writeShort(this.template.id);
            m.writer().writeByte(this.zone_id);
            m.writer().writeByte(this.template.type_view_p);
            m.writer().writeShort(p.x);
            m.writer().writeShort(p.y);
            m.writer().writeInt(p.body.get_hp_max(true));
            m.writer().writeInt(p.hp);
            m.writer().writeInt(p.body.get_mp_max(true));
            m.writer().writeInt(p.mp);
            m.writer().writeByte(this.template.b);
            m.writer().writeByte(this.template.specMap);
            if (this.template.b == 1) {
                m.writer().writeInt(this.template.data[0].length);
                m.writer().write(this.template.data[0]);
                m.writer().writeInt(this.template.data[1].length);
                m.writer().write(this.template.data[1]);
                m.writer().writeByte(this.template.vgos.size());
                for (int i = 0; i < this.template.vgos.size(); i++) {
                    m.writer().writeUTF(this.template.vgos.get(i).map_go[0].template.name);
                    m.writer().writeShort(this.template.vgos.get(i).xold);
                    m.writer().writeShort(this.template.vgos.get(i).yold);
                }
            }
            m.writer().writeByte(this.template.IDBack);
            m.writer().writeShort(this.template.HBack);
            m.writer().writeByte(this.template.id_eff_map);
            m.writer().writeByte(this.template.level);
            m.writer().writeByte(this.template.typeChangeMap);
            if (this.template.specMap == 3) {
                m.writer().writeByte(this.template.mPosMapTrain.length);
                for (int i = 0; i < this.template.mPosMapTrain.length; i++) {
                    for (int j = 0; j < this.template.mPosMapTrain[i].length; j++) {
                        m.writer().writeByte(this.template.mPosMapTrain[i][j]);
                    }
                }
                m.writer().writeUTF(this.template.strTimeChange);
            }
            m.writer().writeUTF(this.template.name);
            p.conn.addmsg(m);
            m.cleanup();
        } catch (IOException ex) {
            Logger.getLogger(Map.class.getName()).log(java.util.logging.Level.SEVERE, null, ex);
        }
    }
